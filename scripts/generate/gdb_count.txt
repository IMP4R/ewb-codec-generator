set pagination off
set architecture i386
handle SIGUSR1 nostop noprint pass
attach PIDHERE
set $p = (int)mmap(0, 0x40000, 3, 0x22, -1, 0)
restore /tmp/inject.bin binary $p
set $inj = $p
set $idx = 0
set $rem = REMHERE
set $dest = 0
set $plen = 0
set $hits = 0
break *0x4a9fdd
commands
silent
set $dest = $edi
set $plen = $esi
set $hits = $hits + 1
continue
end
break *0x4aa02b
commands
silent
if $rem > 0
  if $rem > 1024
    set $n2 = 1024
  else
    set $n2 = $rem
  end
  set $rc = (int)0x54cc10($dest, $inj + $idx, $n2)
  set $idx = $idx + $n2
  set $rem = $rem - $n2
else
  set $n2 = 0
end
set *(int*)$plen = $n2
printf "INJECT block len=%d\n", $n2
continue
end
break *0x4aa0c0
commands
silent
printf "DONE hits=%d idx=%d\n", $hits, $idx
detach
quit
end
continue
