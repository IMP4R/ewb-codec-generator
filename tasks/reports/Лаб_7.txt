Часть 1. Вычисление выражения (2·a − 1)/(b − 2)
Данные и результат — short int (16 бит со знаком). Модуль на Ассемблере вычисляет выражение, программа на Visual C++ выполняет корректный ввод с контролем диапазона и сравнивает результаты.
Модуль на Ассемблере (lab7_1_calc.asm)
; Лабораторная работа 7, часть 1. Вариант 1: y = (2*a - 1) / (b - 2)
; Все данные short int (16 бит со знаком). Модуль вызывается из C++.
.586P
.MODEL FLAT,C

DATA SEGMENT
PUBLIC a, b, res
a DW ?
b DW ?
res DW ?
DATA ENDS

CODE SEGMENT
PUBLIC calc_expr
calc_expr PROC
    MOV  AX, a          ; AX = a
    ADD  AX, AX         ; AX = 2*a
    DEC  AX             ; AX = 2*a - 1
    CWD                 ; DX:AX = знаковое расширение AX
    MOV  BX, b
    SUB  BX, 2          ; BX = b - 2
    IDIV BX             ; AX = частное, DX = остаток
    MOV  res, AX        ; res = (2*a-1)/(b-2)
    RET
calc_expr ENDP
CODE ENDS
END

Программа на Visual C++ (lab7_1_calc.cpp)
// Лабораторная работа 7, часть 1. Вариант 1: y = (2*a - 1) / (b - 2)
// Программа на Visual C++ (главная), вызывает ассемблерный модуль calc_expr.
#include <stdio.h>
#include <conio.h>
#include <limits.h>

extern "C" void calc_expr(void);
extern "C" short int a, b, res;

// Контроль ввода: только целые числа в диапазоне short int
int read_short(const char* name, short int* out)
{
    long v;
    int ok = 0;
    while (!ok)
    {
        printf("Введите %s (целое от %d до %d): ", name, SHRT_MIN, SHRT_MAX);
        if (scanf("%ld", &v) != 1)                 // произвольные символы
        {
            printf("Ошибка: введено не целое число!\n");
            while (getchar() != '\n');             // очистка буфера
            continue;
        }
        if (v < SHRT_MIN || v > SHRT_MAX)
        {
            printf("Ошибка: значение вне диапазона short int!\n");
            continue;
        }
        ok = 1;
    }
    *out = (short int)v;
    return 1;
}

int main()
{
    short int y;
    printf("Вычисление y = (2*a - 1) / (b - 2), a, b : short int\n");
    read_short("a", &a);
    read_short("b", &b);
    if (b == 2)
    {
        printf("Ошибка: деление на ноль (b - 2 = 0)!\n");
        _getch();
        return 1;
    }
    // контроль промежуточного результата 2*a - 1 (16 бит со знаком)
    if (2 * (int)a - 1 > SHRT_MAX || 2 * (int)a - 1 < SHRT_MIN)
    {
        printf("Ошибка: промежуточное значение 2*a - 1 = %d вне диапазона short int!\n", 2 * (int)a - 1);
        _getch();
        return 1;
    }
    y = (short int)((2 * (int)a - 1) / (b - 2));   // эталон на C++
    calc_expr();                                    // расчёт на Ассемблере
    printf("Результат на C++       : y = %d\n", y);
    printf("Результат на Ассемблере: y = %d\n", res);
    if (y != res) printf("ВНИМАНИЕ: результаты не совпали!\n");
    _getch();
    return 0;
}

Пояснения к модулю
ADD AX,AX — удвоение a; DEC AX — вычитание единицы: AX = 2a − 1.
CWD — знаковое расширение AX в пару DX:AX (необходимо для IDIV).
IDIV BX — знаковое деление DX:AX на BX (b − 2); частное остаётся в AX.
Программа на C++ контролирует: ввод только целых чисел (иначе сообщение об ошибке), диапазон short int, b ≠ 2, и диапазон промежуточного значения 2a − 1.
Тестовые проверки

Особые случаи: при b = 2 выводится сообщение о делении на ноль; при a, дающем 2a−1 вне диапазона short int (например a = 20000: 2a−1 = 39999 > 32767), выводится сообщение о выходе промежуточного значения за диапазон; при вводе нечисловых символов программа сообщает об ошибке ввода и запрашивает данные заново.
Часть 2. Логические команды (вариант 1)
Задание: 1) проверить содержимое битов 1, 4, 7; 2) установить в единицу биты 2, 5, 8; 3) проверить биты 3, 6, 9 и обнулить те из них, которые равны 1; 4) выполнить двойную инверсию x.
Модуль на Ассемблере (lab7_2_bits.asm)
; Лабораторная работа 7, часть 2. Вариант 1
; 1) Проверить содержимое битов 1, 4, 7.
; 2) Установить в единицу биты 2, 5, 8.
; 3) Проверить биты 3, 6, 9: если равны 1, обнулить их.
; 4) Сделать двойную инверсию x (после неё значение x не меняется).
.586P
.MODEL FLAT,C

DATA SEGMENT
PUBLIC x, m1_set, m3_set
x        DW ?     ; рабочее число
m1_set   DW ?     ; маска: какие из битов 1,4,7 были установлены
m3_set   DW ?     ; маска: какие из битов 3,6,9 были сброшены
DATA ENDS

CODE SEGMENT
PUBLIC bit_ops
bit_ops PROC
    MOV  AX, x
    ; --- 1) проверка битов 1,4,7 (маска 0092h = 0000 0001 0010 0100b... см. ниже)
    MOV  WORD PTR m1_set, 0
    TEST AX, 0092h          ; биты 1(02h), 4(10h), 7(80h)
    JZ   m1_done
    MOV  BX, AX
    AND  BX, 0092h
    MOV  m1_set, BX
m1_done:
    ; --- 2) установить биты 2,5,8 (маска 0124h = 04h+20h+100h)
    OR   AX, 0124h
    ; --- 3) биты 3,6,9 (маска 0248h = 08h+40h+200h)
    MOV  WORD PTR m3_set, 0
    MOV  BX, AX
    AND  BX, 0248h
    JZ   m3_done
    MOV  m3_set, BX         ; что будет сброшено
    NOT  BX                 ; инверсная маска: 1 = оставить
    AND  AX, BX
m3_done:
    ; --- 4) двойная инверсия: NOT NOT x = x, значение не меняется
    NOT  AX
    NOT  AX
    MOV  x, AX
    RET
bit_ops ENDP
CODE ENDS
END

Программа на Visual C++ (lab7_2_bits.cpp)
// Лабораторная работа 7, часть 2. Вариант 1: операции над битами 16-разрядного x
#include <stdio.h>
#include <conio.h>
#include <string.h>
#include <stdlib.h>

extern "C" void bit_ops(void);
extern "C" unsigned short int x;
extern "C" unsigned short int m1_set, m3_set;

void to_bin(unsigned v, char* buf)
{
    for (int i = 15; i >= 0; --i) *buf++ = (v >> i) & 1 ? '1' : '0';
    *buf = 0;
}

// ввод 16-разрядного числа в двоичной форме (с проверкой)
int read_bin(const char* name, unsigned short* out)
{
    char s[64];
    while (1)
    {
        printf("Введите %s (16 двоичных разрядов, например 1011001010101111): ", name);
        if (scanf("%63s", s) != 1) return 0;
        size_t n = strlen(s);
        if (n == 0 || n > 16) { printf("Ошибка: нужно от 1 до 16 символов 0/1\n"); continue; }
        unsigned v = 0; int ok = 1;
        for (size_t i = 0; i < n; ++i)
        {
            if (s[i] != '0' && s[i] != '1') { ok = 0; break; }
            v = (v << 1) | (unsigned)(s[i] - '0');
        }
        if (!ok) { printf("Ошибка: допустимы только символы 0 и 1\n"); continue; }
        *out = (unsigned short)v;
        return 1;
    }
}

int main()
{
    char buf[20];
    printf("Лабораторная работа 7, часть 2. Вариант 1\n");
    printf("1) проверка битов 1,4,7  2) установка битов 2,5,8  3) сброс битов 3,6,9 если=1  4) двойная инверсия\n\n");
    read_bin("x", &x);
    to_bin(x, buf);
    printf("Исходный x  = %s = %u (десятичное)\n", buf, x);

    bit_ops();   // операции выполняются модулем на Ассемблере

    printf("Биты 1,4,7, которые были равны 1: ");
    if (m1_set == 0) printf("нет\n");
    else { for (int i = 0; i < 16; ++i) if (m1_set & (1 << i)) printf("%d ", i); printf("\n"); }
    printf("Биты 3,6,9, которые были сброшены: ");
    if (m3_set == 0) printf("нет\n");
    else { for (int i = 0; i < 16; ++i) if (m3_set & (1 << i)) printf("%d ", i); printf("\n"); }

    to_bin(x, buf);
    printf("Итоговый x  = %s = %u (десятичное)\n", buf, x);
    printf("Двойная инверсия не изменила значение (NOT NOT x = x).\n");
    _getch();
    return 0;
}

Используемые маски

Сброс выполняется парой NOT/AND: в инверсной маске единицы стоят на позициях, которые нужно сохранить, поэтому AND обнуляет только установленные биты из проверяемых. Двойная инверсия NOT NOT x возвращает исходное значение — это инволюция.
Пример выполнения (x = 1011001010101111b = 45743)

Значение x после двойной инверсии не изменилось, что подтверждает корректность операции.
